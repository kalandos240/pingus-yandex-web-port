<body>
  <main id="game-shell">
    <canvas id="canvas" tabindex="0" oncontextmenu="event.preventDefault()" style="width: 780px; height: 390px;"></canvas>
    <section id="loading" aria-live="polite">
      <div id="loading-card">
        <h1 id="loading-title">Pingus</h1>
        <div id="loading-status">Loading…</div>
        <progress id="progress" max="100" value="100"></progress>
      </div>
    </section>
  </main>

  <script>
    (() => {
      const nativeCanvasGetContext = HTMLCanvasElement.prototype.getContext;
      HTMLCanvasElement.prototype.getContext = function(type, options) {
        if (type === '2d') {
          const readbackOptions = options && typeof options === 'object'
            ? { ...options, willReadFrequently: true }
            : { willReadFrequently: true };
          return nativeCanvasGetContext.call(this, type, readbackOptions);
        }
        return nativeCanvasGetContext.apply(this, arguments);
      };

      const loading = document.getElementById('loading');
      const status = document.getElementById('loading-status');
      const progress = document.getElementById('progress');
      const canvas = document.getElementById('canvas');
      let gameReadySent = false;
      let saveInProgress = false;
      let saveRequestedAgain = false;
      let autosaveTimer = 0;

      // Before the first rendered frame we must not gate startup on focus: an
      // embedded Yandex iframe may not own focus yet. After the game is ready,
      // losing focus pauses both the native simulation and audio until focus
      // returns. document.hidden remains an unconditional pause condition.
      window.pingusPagePaused = () =>
        document.hidden || (gameReadySent && !document.hasFocus());

      const text = {
        en: { loading: 'Loading…', failed: 'The game could not be started.' },
        ru: { loading: 'Загрузка…', failed: 'Не удалось запустить игру.' }
      };
      let uiLanguage = 'en';

      const normalizeLanguage = (value) => {
        const code = String(value || '').toLowerCase().split(/[-_]/)[0];
        return code === 'ru' ? 'ru' : 'en';
      };

      const applyLanguage = (value) => {
        uiLanguage = normalizeLanguage(value);
        document.documentElement.lang = uiLanguage;
        if (!gameReadySent) status.textContent = text[uiLanguage].loading;
        try {
          const locale = uiLanguage === 'ru' ? 'ru_RU.UTF-8' : 'en_US.UTF-8';
          ENV.LANG = locale;
          ENV.LC_ALL = locale;
        } catch (_) {}
      };

      const setLoadingError = (message) => {
        const errorText = message || text[uiLanguage].failed;
        document.documentElement.dataset.pingusError = errorText;
        loading.hidden = false;
        status.textContent = errorText;
        progress.removeAttribute('value');
      };

      window.yandexSDKPromise = (async () => {
        try {
          if (typeof YaGames === 'undefined') return null;
          const ysdk = await YaGames.init();
          window.ysdk = ysdk;
          applyLanguage(ysdk?.environment?.i18n?.lang);
          return ysdk;
        } catch (error) {
          console.warn('Yandex Games SDK initialization failed:', error);
          return null;
        }
      })();

      const fitCanvas = () => {
        const logicalWidth = Math.max(1, canvas.width || 800);
        const logicalHeight = Math.max(1, canvas.height || 600);
        const scale = Math.min(window.innerWidth / logicalWidth, window.innerHeight / logicalHeight);
        canvas.style.width = `${Math.max(1, Math.floor(logicalWidth * scale))}px`;
        canvas.style.height = `${Math.max(1, Math.floor(logicalHeight * scale))}px`;
      };
      window.addEventListener('resize', fitCanvas, { passive: true });
      new MutationObserver(fitCanvas).observe(canvas, { attributes: true, attributeFilter: ['width', 'height'] });
      fitCanvas();

      const getAudioContext = () =>
        window.SDL?.audioContext || window.SDL2?.audioContext || window.AL?.currentContext?.audioCtx || null;

      const setAudioPaused = (paused) => {
        const audioContext = getAudioContext();
        if (audioContext) {
          const action = paused ? audioContext.suspend?.() : audioContext.resume?.();
          action?.catch?.(() => {});
        }

        // SDL1_mixer in Emscripten may play OGG music/channels through regular
        // HTMLAudioElement objects, not through the AudioContext above. Pause
        // only media that was actually playing, then resume exactly those.
        const media = [];
        const music = window.SDL?.music?.audio;
        if (music) media.push(music);
        const channels = window.SDL?.channels;
        if (Array.isArray(channels)) {
          for (const channel of channels) {
            if (channel?.audio) media.push(channel.audio);
          }
        }
        for (const audio of new Set(media)) {
          if (paused) {
            if (!audio.paused) {
              audio.__pingusResumeAfterPause = true;
              audio.pause();
            }
          } else if (audio.__pingusResumeAfterPause) {
            audio.__pingusResumeAfterPause = false;
            const play = audio.play?.();
            play?.catch?.(() => {});
          }
        }
      };

      window.pingusSaveNow = () => {
        if (typeof FS === 'undefined' || typeof FS.syncfs !== 'function') return;
        if (saveInProgress) {
          saveRequestedAgain = true;
          return;
        }
        saveInProgress = true;
        FS.syncfs(false, (error) => {
          saveInProgress = false;
          if (error) console.warn('Pingus save synchronization failed:', error);
          if (saveRequestedAgain) {
            saveRequestedAgain = false;
            window.pingusSaveNow();
          }
        });
      };

      window.pingusMarkReady = async () => {
        if (gameReadySent) return;
        gameReadySent = true;
        document.documentElement.dataset.pingusReady = '1';
        fitCanvas();
        loading.hidden = true;
        canvas.focus({ preventScroll: true });
        if (!autosaveTimer) autosaveTimer = window.setInterval(window.pingusSaveNow, 15000);
        const ysdk = await window.yandexSDKPromise;
        try { ysdk?.features?.LoadingAPI?.ready(); } catch (error) { console.warn(error); }
      };

      const syncPagePause = () => {
        const paused = window.pingusPagePaused();
        setAudioPaused(paused);
        if (paused) window.pingusSaveNow();
      };
      document.addEventListener('visibilitychange', syncPagePause);
      window.addEventListener('blur', syncPagePause);
      window.addEventListener('focus', syncPagePause);
      window.addEventListener('pagehide', window.pingusSaveNow);
      window.addEventListener('beforeunload', window.pingusSaveNow);
      window.addEventListener('error', (event) => {
        if (!gameReadySent) setLoadingError(event.message);
      });
      window.addEventListener('unhandledrejection', (event) => {
        if (!gameReadySent) setLoadingError(String(event.reason || text[uiLanguage].failed));
      });

      const unlockAudio = () => setAudioPaused(false);
      window.addEventListener('pointerdown', unlockAudio, { once: true, passive: true });
      window.addEventListener('keydown', unlockAudio, { once: true });
      document.addEventListener('keydown', (event) => {
        if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space'].includes(event.code)) {
          event.preventDefault();
        }
      }, { passive: false });

      applyLanguage(navigator.language);

      window.Module = {
        canvas,
        noAudioDecoding: true,
        preRun: [() => {
          // Guard legacy Pingus saves against Emscripten chmod(..., 0).
          if (!FS.__pingusChmodPatched) {
            const nativeChmod = FS.chmod.bind(FS);
            FS.chmod = (path, mode) => {
              const name = String(path || '');
              if (name.startsWith('/home/web_user/') && (mode & 0o777) === 0) mode = 0o600;
              return nativeChmod(path, mode);
            };
            FS.__pingusChmodPatched = true;
          }
          try { FS.mkdir('/home'); } catch (_) {}
          try { FS.mkdir('/home/web_user'); } catch (_) {}
          try { ENV.HOME = '/home/web_user'; } catch (_) {}

          addRunDependency('pingus-locale');
          Promise.race([
            window.yandexSDKPromise,
            new Promise((resolve) => window.setTimeout(() => resolve(null), 4000))
          ]).finally(() => removeRunDependency('pingus-locale'));

          addRunDependency('pingus-idbfs');
          try {
            FS.mount(IDBFS, {}, '/home/web_user');
            FS.syncfs(true, (error) => {
              if (error) console.warn('Pingus save restore failed:', error);
              removeRunDependency('pingus-idbfs');
            });
          } catch (error) {
            console.warn('IDBFS unavailable:', error);
            removeRunDependency('pingus-idbfs');
          }
        }],
        print: (message) => console.log(message),
        printErr: (message) => console.error(message),
        onAbort: (reason) => setLoadingError(`Pingus stopped: ${reason || 'unknown error'}`),
        setStatus: (message) => {
          if (!message) return;
          status.textContent = message;
          const match = /\((\d+)\/(\d+)\)/.exec(message);
          if (match) {
            progress.max = Number(match[2]);
            progress.value = Number(match[1]);
          }
        },
        monitorRunDependencies: (left) => {
          if (!left) progress.value = progress.max;
        }
      };
    })();
  </script>
  <script src="pingus.js"></script>


</body></html>
